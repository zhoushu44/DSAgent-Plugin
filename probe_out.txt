224:   // ── 动态推荐：规则引擎 ──
225:   /** 从最近几轮 prompt/response 文本推断「平台 × 已完成阶段」。 */
226:   function inferContext(items: readonly { readonly prompt: string; readonly response: string }[]): { platform?: string; stage?: Stage } {
227:     // 取最近 3 轮，从最新往回扫：最新一轮命中的才算数（话题会切换，
228:     // 旧轮次的动作词不该覆盖新话题；如「查看账号连接状态」的「账号」
229:     // 不该把之后的「分析商品评价」阶段抢成 manage）。
230:     const recent = items.slice(-3)
231:     if (recent.length === 0) return {}
232: 
233:     // 平台：逐轮（新→旧）在 prompt+response 里找；助手回复也常提平台名
234:     // （如「已采集抖音视频 50 条」），可作辅助。轮内 prompt 优先于
235:     // response——用户说的是主话题，回复里顺带提到其他平台（如「也可
236:     // 试试闲鱼」）不该抢走平台；同段内词表序先命中先得。
237:     let platform: string | undefined
238:     for (let i = recent.length - 1; i >= 0 && !platform; i--) {
239:       const { prompt, response } = recent[i]
240:       for (const segment of [prompt, response]) {
241:         for (const entry of PLATFORM_WORDS) {
242:           if (entry.words.some((w) => segment.includes(w))) {
243:             platform = entry.platform
244:             break
245:           }
246:         }
247:         if (platform) break
248:       }
249:     }
250: 
251:     // 阶段：逐轮（新→旧）只看用户 prompt——助手回复常含「账号 xxx」
252:     // 「分析完成」等系统噪声，会把刚完成的动作误判成 manage/analyze（实测踩坑）。
253:     // 轮内取词表序最后一个命中（collect→analyze→publish→manage，越靠后越是刚完成的动作）。
254:     let stage: Stage | undefined
255:     for (let i = recent.length - 1; i >= 0 && !stage; i--) {
256:       const prompt = recent[i].prompt
257:       for (const entry of STAGE_WORDS) {
258:         if (entry.words.some((w) => prompt.includes(w))) stage = entry.stage
259:       }
260:     }
