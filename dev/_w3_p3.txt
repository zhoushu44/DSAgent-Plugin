
/** 扫描 text 里所有以 prefix 开头的标识符（只取 prefix 之后的连续字符）。 */
export function scanPrefixed(text, prefix, charSet = ID_CHARS) {
  const out = []
  let i = text.indexOf(prefix)
  while (i >= 0) {
    let j = i + prefix.length
    while (j < text.length && charSet.includes(text[j])) j++
    const token = text.slice(i, j)
    if (token.length > prefix.length) out.push(token)
    i = text.indexOf(prefix, j)
  }
  return uniq(out)
}

/** 从 src 源码文本集合里抽「已注册工具名」。 */
export function collectRegisteredTools(srcTexts) {
  const out = new Set()
  for (const text of srcTexts) {
    for (const t of scanPrefixed(text, 'dsagent_')) out.add(t)
  }
  return out
}

/** 从技能正文抽「提到的工具名」与「引用的锚点集路径」。 */
export function extractSkillRefs(text) {
  return {
    tools: scanPrefixed(text, 'dsagent_'),
    configs: scanPrefixed(text, 'publish-config/', PATH_CHARS),
  }
}

/** 读 src 下全部 .ts 文本（跳过 node_modules / lib / 点目录）。 */
export function readSrcTexts(root = REPO_ROOT) {
  const out = []
  const walk = (dir) => {
    let entries = []
    try {
      entries = fs.readdirSync(dir, { withFileTypes: true })
    } catch {
      return
    }
    for (const e of entries) {
      if (e.name === 'node_modules' || e.name === 'lib' || e.name.startsWith('.')) continue
      const p = path.join(dir, e.name)
      if (e.isDirectory()) walk(p)
      else if (e.name.endsWith('.ts')) out.push(fs.readFileSync(p, 'utf8'))
    }
  }
  walk(path.join(root, 'src'))
  return out
}
