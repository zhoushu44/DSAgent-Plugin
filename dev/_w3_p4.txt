
/**
 * 单平台一致性判定（纯函数，便于 --selftest 用合成样例跑）。
 *
 * @param {{id: string, text: string}} skill
 * @param {object} spec PLATFORMS 里的一项
 * @param {{srcText: string, registered: Set<string>, configExists: (p: string) => boolean}} env
 */
export function analyzeSkill(skill, spec, env) {
  const text = stripBom(skill.text)
  const issues = []
  const push = (severity, code, message) => issues.push({ severity, code, message })

  const { tools, configs } = extractSkillRefs(text)
  const publishTools = tools.filter((t) => t.includes('publish'))

  // 1) 技能声明的发布工具是否在宿主真实注册
  if (!publishTools.length) {
    push('error', 'no_publish_tool', '正文没有提到任何 dsagent_*publish* 工具，模型无从调用')
  } else {
    const implemented = publishTools.filter((t) => env.registered.has(t))
    const missing = publishTools.filter((t) => !env.registered.has(t))
    if (!implemented.length) {
      if (hasAny(text, DOWNGRADE_WORDS)) {
        push(
          'warn',
          'downgrade_only',
          '工具 ' + missing.join(' / ') + ' 在 src/ 未注册；技能已给「降级为指令型」路径，属已知缺口',
        )
      } else {
        push(
          'error',
          'tool_missing',
          '技能声明的发布工具 ' + missing.join(' / ') + ' 在 src/ 一个都没注册，且未给降级路径',
        )
      }
    }
  }

  // 2) 技能声明的自家发布域名是否在 src/ 有实现证据
  const declared = spec.domains.filter((d) => text.includes(d))
  if (!declared.length) {
    push('warn', 'domain_not_declared', '正文没出现自家平台域名（' + spec.domains.join(' / ') + '）')
  } else if (!declared.some((d) => env.srcText.includes(d))) {
    push(
      'gap',
      'domain_not_in_src',
      '技能声明域名 ' + declared.join(' / ') + ' 在 src/ 无任何实现证据（该平台发布链路尚未落地）',
    )
  }

  // 3) 锚点集：技能引用了就必须存在；本平台应引用却没引用则提醒
  for (const cfg of configs) {
    if (!env.configExists(cfg)) push('error', 'config_missing', '技能引用的锚点集 ' + cfg + ' 不存在')
  }
  if (!configs.length && spec.configs.length) {
    push('warn', 'config_not_referenced', '技能未引用锚点集 ' + spec.configs.join(' / ') + '（改锚点无出处）')
  }

  return { id: skill.id, label: spec.label, issues }
}

export const SEVERITY_RANK = { error: 3, gap: 2, warn: 1 }

/** 汇总所有平台。 */
export function runCheck(options = {}) {
  const root = options.root || REPO_ROOT
  const skillsDir = options.dir || path.join(root, 'skills')
  const onlyIds = options.ids || null
  const srcTexts = options.srcTexts || readSrcTexts(root)
  const srcText = srcTexts.join(LF)
  const registered = options.registered || collectRegisteredTools(srcTexts)
  const env = {
    srcText,
    registered,
    configExists: (p) => fs.existsSync(path.join(root, p)),
  }

  const reports = []
  const missingSkills = []
  for (const spec of PLATFORMS) {
    if (onlyIds && !onlyIds.includes(spec.id)) continue
    const file = path.join(skillsDir, spec.id, 'SKILL.md')
    if (!fs.existsSync(file)) {
      missingSkills.push(spec.id)
      continue
    }
    reports.push(analyzeSkill({ id: spec.id, text: fs.readFileSync(file, 'utf8') }, spec, env))
  }

  const count = (sev) => reports.reduce((n, r) => n + r.issues.filter((i) => i.severity === sev).length, 0)
  return {
    reports,
    missingSkills,
    errorCount: count('error'),
    gapCount: count('gap'),
    warnCount: count('warn'),
  }
}
